Table role {
  id int [pk, increment]
  name varchar(50) [unique, not null]
}

Table department {
  id int [pk, increment]
  name varchar(100) [not null]
}

Table building {
  id int [pk, increment]
  name varchar(100) [not null]
}

Table floor {
  id int [pk, increment]
  building_id int [not null]
  name varchar(100) [not null]
}

Table resource {
  id int [pk, increment]
  name varchar(100) [not null]
  type varchar(20) [not null]
  department_id int
  floor_id int [not null]
  capacity int [not null]
  desk_mode varchar(20) [not null]
}

Table resource_features {
  resource_id int [not null]
  feature varchar(255) [not null]
}

Table user_account {
  id int [pk, increment]
  username varchar(100) [unique, not null]
  password varchar(255) [not null]
  role_id int
  department_id int
  email varchar(100) [unique, not null]
}

Table booking {
  id int [pk, increment]
  resource_id int [not null]
  user_id int [not null]
  start_utc timestamp [not null]
  end_utc timestamp [not null]
  status varchar(30) [not null]
  recurrence_group_id uuid
}

Table waitlist_entry {
  id int [pk, increment]
  resource_id int [not null]
  start_utc timestamp [not null]
  end_utc timestamp [not null]
  user_id int [not null]
  created_at_utc timestamp [not null]
}

Table waitlist_offer {
  id int [pk, increment]
  waitlist_entry_id int [not null]
  status varchar(30) [not null]
  offered_at_utc timestamp [not null]
  expires_at_utc timestamp [not null]
  provisional_booking_id int [unique]
}

Table desk_assignment {
  id int [pk, increment]
  resource_id int
  user_id int
  start_utc timestamp [not null]
  end_utc timestamp
}

Table notification {
  id int [pk, increment]
  user_id int
  message varchar [not null]
  type varchar
  read_status boolean [default: false]
  created_at timestamp
}
